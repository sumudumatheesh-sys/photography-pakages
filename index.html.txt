  if (!/^\d{7,15}$/.test(digits)) { setError(form.phone, 'Please enter a valid phone number.'); valid = false; }
  else setError(form.phone, '');

  if (!valid) return;

  document.getElementById('formSuccess').classList.add('show');
  form.reset();
  setTimeout(() => document.getElementById('formSuccess').classList.remove('show'), 6000);
});

  form[id].addEventListener('input', () => setError(form[id], '')));
